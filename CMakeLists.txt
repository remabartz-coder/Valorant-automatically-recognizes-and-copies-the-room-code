cmake_minimum_required(VERSION 3.24)
project(QueuePilot VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
if(NOT WIN32)
  message(FATAL_ERROR "QueuePilot requires Windows 10 1903 or newer")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "QueuePilot requires an x64 build. Configure with -A x64.")
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network)
set(FFMPEG_ROOT "${CMAKE_SOURCE_DIR}/.deps/ffmpeg" CACHE PATH "FFmpeg SDK")
set(ONNXRUNTIME_ROOT "${CMAKE_SOURCE_DIR}/.deps/onnxruntime" CACHE PATH "ONNX Runtime SDK")
find_path(FFMPEG_INCLUDE libavformat/avformat.h HINTS "${FFMPEG_ROOT}/include" REQUIRED)
foreach(component avformat avcodec avutil swscale)
  find_library(${component}_LIB NAMES ${component} HINTS "${FFMPEG_ROOT}/lib" REQUIRED)
endforeach()
find_path(ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" REQUIRED)
find_library(ORT_LIB onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" REQUIRED)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive- /EHsc)
endif()
add_library(pilot_core src/code_gate.cpp src/ocr_engine.cpp src/code_locator.cpp)
target_include_directories(pilot_core PUBLIC src PRIVATE "${ORT_INCLUDE}")
target_link_libraries(pilot_core PUBLIC Qt6::Core Qt6::Gui PRIVATE "${ORT_LIB}")
add_library(pilot_sources src/video_decoder.cpp src/stream_resolver.cpp src/window_capture.cpp)
target_include_directories(pilot_sources PUBLIC src PRIVATE "${FFMPEG_INCLUDE}")
target_link_libraries(pilot_sources PUBLIC Qt6::Core Qt6::Gui Qt6::Network PRIVATE
  "${avformat_LIB}" "${avcodec_LIB}" "${avutil_LIB}" "${swscale_LIB}"
  windowsapp d3d11 dxgi user32 ole32)
add_library(pilot_input src/game_input.cpp)
target_include_directories(pilot_input PUBLIC src)
target_link_libraries(pilot_input PUBLIC Qt6::Core PRIVATE user32)
add_library(pilot_notifications src/speech_notifier.cpp)
target_include_directories(pilot_notifications PUBLIC src)
target_link_libraries(pilot_notifications PRIVATE sapi ole32)
add_executable(QueuePilot WIN32 src/main.cpp src/main_window.cpp src/preview_widget.cpp src/recognition_worker.cpp
  src/app_resources.qrc)
configure_file(src/queuepilot.rc.in "${CMAKE_CURRENT_BINARY_DIR}/queuepilot.rc" @ONLY)
target_sources(QueuePilot PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/queuepilot.rc")
target_link_libraries(QueuePilot PRIVATE pilot_core pilot_sources pilot_input pilot_notifications Qt6::Widgets)
target_compile_definitions(QueuePilot PRIVATE QUEUEPILOT_VERSION="${PROJECT_VERSION}")
include(CTest)
if(BUILD_TESTING)
add_executable(core_tests tests/core_tests.cpp src/code_gate.cpp)
target_include_directories(core_tests PRIVATE src)
target_link_libraries(core_tests PRIVATE Qt6::Core Qt6::Gui)
add_test(NAME code_gate COMMAND core_tests)
add_executable(resolver_tests tests/resolver_tests.cpp src/stream_resolver.cpp)
target_include_directories(resolver_tests PRIVATE src)
target_link_libraries(resolver_tests PRIVATE Qt6::Core Qt6::Network)
add_test(NAME resolver COMMAND resolver_tests)
add_executable(input_tests tests/input_tests.cpp)
target_link_libraries(input_tests PRIVATE pilot_input)
add_test(NAME input_guards COMMAND input_tests)
add_executable(ocr_smoke tests/ocr_smoke.cpp)
target_link_libraries(ocr_smoke PRIVATE pilot_core)
add_executable(ocr_probe tests/ocr_probe.cpp)
target_link_libraries(ocr_probe PRIVATE pilot_core)
add_test(NAME ocr_model COMMAND ocr_smoke "${CMAKE_SOURCE_DIR}/models/recognizer.onnx" "${CMAKE_SOURCE_DIR}/models/keys.txt")
add_executable(pipeline_tests tests/pipeline_tests.cpp src/recognition_worker.cpp)
target_link_libraries(pipeline_tests PRIVATE pilot_core pilot_sources)
add_test(NAME pipeline COMMAND pipeline_tests "${CMAKE_SOURCE_DIR}/models")
add_executable(real_video_tests tests/real_video_tests.cpp src/recognition_worker.cpp)
target_link_libraries(real_video_tests PRIVATE pilot_core)
set(QUEUEPILOT_REAL_VIDEO_FRAMES "${CMAKE_SOURCE_DIR}/test-output/real-kog460/frames-4fps"
    CACHE PATH "Optional fps=4 frames from the user-provided test recording")
if(EXISTS "${QUEUEPILOT_REAL_VIDEO_FRAMES}/frame-157.png")
  add_test(NAME real_video COMMAND real_video_tests "${CMAKE_SOURCE_DIR}/models"
    "${QUEUEPILOT_REAL_VIDEO_FRAMES}" "${CMAKE_SOURCE_DIR}/test-output/real-kog460/recognition-report.json")
  set_tests_properties(real_video PROPERTIES TIMEOUT 70)
endif()
add_executable(locator_smoke tests/locator_smoke.cpp)
target_link_libraries(locator_smoke PRIVATE pilot_core)
add_test(NAME code_locator COMMAND locator_smoke "${CMAKE_SOURCE_DIR}/models")
set_tests_properties(ocr_model pipeline code_locator PROPERTIES TIMEOUT 30)
add_executable(ui_smoke tests/ui_smoke.cpp src/main_window.cpp src/preview_widget.cpp src/recognition_worker.cpp)
target_link_libraries(ui_smoke PRIVATE pilot_core pilot_sources pilot_input pilot_notifications Qt6::Widgets)
set_target_properties(ui_smoke PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/ui-test")
add_test(NAME ui_workflow COMMAND ui_smoke)
set_tests_properties(ui_workflow PROPERTIES TIMEOUT 20)
add_executable(preview_tests tests/preview_tests.cpp src/preview_widget.cpp)
target_include_directories(preview_tests PRIVATE src)
target_link_libraries(preview_tests PRIVATE Qt6::Widgets)
add_test(NAME preview_selection COMMAND preview_tests)
set_tests_properties(preview_selection PROPERTIES TIMEOUT 10)
add_executable(speech_tests tests/speech_tests.cpp)
target_link_libraries(speech_tests PRIVATE pilot_notifications)
add_test(NAME speech_notifications COMMAND speech_tests)
set_tests_properties(speech_notifications PROPERTIES TIMEOUT 10)
add_executable(decoder_smoke tests/decoder_smoke.cpp)
target_link_libraries(decoder_smoke PRIVATE pilot_sources)
add_executable(capture_smoke tests/capture_smoke.cpp)
target_link_libraries(capture_smoke PRIVATE pilot_sources Qt6::Widgets)
find_program(FFMPEG_EXECUTABLE NAMES ffmpeg.exe HINTS "${FFMPEG_ROOT}/bin" REQUIRED)
add_test(NAME decoder_fixture COMMAND "${FFMPEG_EXECUTABLE}" -hide_banner -loglevel error -y
  -f lavfi -i testsrc=size=320x180:rate=30 -t 2 -c:v mpeg4 -q:v 3 "${CMAKE_BINARY_DIR}/decoder-fixture.mp4")
set_tests_properties(decoder_fixture PROPERTIES FIXTURES_SETUP decoder_video TIMEOUT 15)
add_test(NAME decoder COMMAND decoder_smoke "${CMAKE_BINARY_DIR}/decoder-fixture.mp4")
set_tests_properties(decoder PROPERTIES FIXTURES_REQUIRED decoder_video TIMEOUT 15)
add_test(NAME window_capture COMMAND capture_smoke)
set_tests_properties(window_capture PROPERTIES TIMEOUT 15 LABELS desktop)
set_tests_properties(ui_workflow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=windows")
set_tests_properties(preview_selection PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
file(GLOB FF_RUNTIME "${FFMPEG_ROOT}/bin/avcodec-*.dll" "${FFMPEG_ROOT}/bin/avformat-*.dll"
  "${FFMPEG_ROOT}/bin/avutil-*.dll" "${FFMPEG_ROOT}/bin/swscale-*.dll" "${FFMPEG_ROOT}/bin/swresample-*.dll")
file(GLOB ORT_RUNTIME "${ONNXRUNTIME_ROOT}/lib/*.dll")
foreach(target QueuePilot core_tests resolver_tests input_tests ocr_smoke ocr_probe pipeline_tests real_video_tests locator_smoke ui_smoke preview_tests speech_tests decoder_smoke capture_smoke)
  if(NOT TARGET ${target})
    continue()
  endif()
  foreach(dll IN LISTS FF_RUNTIME ORT_RUNTIME)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${dll}" "$<TARGET_FILE_DIR:${target}>")
  endforeach()
  foreach(component Core Gui Widgets Network)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "$<TARGET_FILE:Qt6::${component}>" "$<TARGET_FILE_DIR:${target}>")
  endforeach()
endforeach()
foreach(target QueuePilot ui_smoke)
  if(NOT TARGET ${target})
    continue()
  endif()
  add_custom_command(TARGET ${target} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/models" "$<TARGET_FILE_DIR:${target}>/models"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${Qt6_DIR}/../../../plugins/tls" "$<TARGET_FILE_DIR:${target}>/tls")
endforeach()
foreach(target QueuePilot ui_smoke ocr_smoke ocr_probe pipeline_tests locator_smoke preview_tests capture_smoke)
  if(NOT TARGET ${target})
    continue()
  endif()
  foreach(plugin QWindowsIntegrationPlugin QOffscreenIntegrationPlugin)
    if(NOT TARGET Qt6::${plugin})
      message(FATAL_ERROR "The Qt platform plugin Qt6::${plugin} is required")
    endif()
    add_custom_command(TARGET ${target} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:${target}>/platforms"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "$<TARGET_FILE:Qt6::${plugin}>" "$<TARGET_FILE_DIR:${target}>/platforms")
  endforeach()
endforeach()
